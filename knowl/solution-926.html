\(\displaystyle{ \begin{aligned}[t] {{\frac{1}{5}}} - {{\frac{1}{9}}} \amp = \frac{1 \cdot 9}{5 \cdot 9}-\frac{1 \cdot 5}{9 \cdot 5} \\ \amp = \frac{9}{45} - \frac{5}{45} \\ \amp = \frac{9-5}{45} \\ \amp = {{\frac{4}{45}}} \end{aligned} }\)
