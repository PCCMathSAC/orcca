\begin{align*} 300A+400B\amp=600\\ 300A+400\left(\substitute{\frac{5}{4}}\right)\amp=600\\ 300A+500\amp=600\\ 300A\amp=100\\ \divideunder{300A}{300}\amp=\divideunder{100}{300}\\ A\amp=\frac{1}{3} \end{align*}
